🔧 Modelos Pydantic (schemas/project.py)

from pydantic import BaseModel, Field
from enum import Enum
from uuid import uuid4

class ShotStatus(str, Enum):
  pending = "pending"
  image_generated = "image_generated"
  clip_generated = "clip_generated"
  scene_rendered = "scene_rendered"
  failed = "failed"

class BaseAsset(BaseModel):
  id: str = Field(default_factory=lambda: str(uuid4()))
  name: str
  include_in_prompt: bool = True
  image_url: str | None = None

class CharacterAsset(BaseAsset):
  type: str = "character"
  physical: str
  clothing: str
  backstory: str

class LocationAsset(BaseAsset):
  type: str = "location"
  prompt: str
  time_of_day: str

class PropAsset(BaseAsset):
  type: str = "prop"
  prompt: str

class Shot(BaseModel):
  id: str = Field(default_factory=lambda: str(uuid4()))
  shotNumber: str
  title: str
  prompt: str
  camera: str = "static"
  linked_asset_ids: list[str] = []
  audio_description: str = ""
  transition_prompt: str = "" # prompt para shot_i → shot_i+1
  status: ShotStatus = ShotStatus.pending
  image_url: str | None = None
  clip_url: str | None = None

class StoryboardScene(BaseModel):
  sceneId: str
  sceneNumber: str
  sceneTitle: str
  shots: list[Shot] = []

class ScriptScene(BaseModel):
  id: str = Field(default_factory=lambda: str(uuid4()))
  slugline: str
  description: str
  time_of_day: str

class Dialogue(BaseModel):
  id: str = Field(default_factory=lambda: str(uuid4()))
  scene_id: str
  character: str
  text: str
  emotion: str = "neutral"

class RenderConfig(BaseModel):
  output_path: str = ""
  duration: float = 0.0
  fps: int = 24
  resolution: str = "1920x1080"
  created_at: str = ""

class StoryboardProject(BaseModel):
  version: str = "1.0"
  projectTitle: str
  globalStyle: str = "Realistic"
  script: dict # {"scenes": [...], "dialogues": [...]}
  assets: list[BaseAsset] = []
  storyboard: dict = {} # {"scenes": [StoryboardScene, ...]}
  render: RenderConfig | None = None

🐛 Problemas del original — Soluciones V1

#Problema originalSolución V1
1 Guion en Markdown con HTML comments frágiles Parser robusto → scenes[] + dialogues[] estructurados
2 Sin TTS ni narración TTS batch por escena/diálogo + mux final
3 No genera video final, solo imágenes Pipeline completo: images → transition clips → scene video → final mux
4 Sin control de estado (pending/generated/failed) ShotStatus enum + tracking por shot/escena
5 supportingImages vacíos, sin versionado image_url en assets + imageHistory por shot
6 motionDescription texto libre Camera enum (pan_left, zoom_in, static, 360_track...) + prompt builder
7 Sin metadatos de render RenderConfig con fps, resolution, duration, outputPath
